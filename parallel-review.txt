+package io.github.payal1762005.minisql.operators;
+
+import io.github.payal1762005.minisql.model.Row;
+
+import java.math.BigDecimal;
+import java.util.Objects;
+
+abstract class AbstractNumericAggregateState implements AggregateState {
+    private final String column;
+    protected long count;
+
+    AbstractNumericAggregateState(String column) {
+        this.column = Objects.requireNonNull(column, "Aggregate input column must not be null");
+    }
+
+    @Override
+    public final void add(Row row) {
+        Objects.requireNonNull(row, "Child operator returned a null row");
+        if (!row.values().containsKey(column)) {
+            throw new IllegalArgumentException(
+                    "Aggregate input column '" + column + "' is missing from row");
+        }
+        String value = row.values().get(column);
+        if (value == null) {
+            return;
+        }
+        addValue(parseNumber(value));
+        count++;
+    }
+
+    @Override
+    public final void merge(AggregateState other) {
+        if (other == null || getClass() != other.getClass()) {
+            throw incompatibleState();
+        }
+        AbstractNumericAggregateState state = (AbstractNumericAggregateState) other;
+        if (!column.equals(state.column)) {
+            throw incompatibleState();
+        }
+        mergeValues(state);
+        count += state.count;
+    }
+
+    protected abstract void addValue(BigDecimal value);
+
+    protected abstract void mergeValues(AbstractNumericAggregateState other);
+
+    protected final String column() {
+        return column;
+    }
+
+    protected final boolean isEmpty() {
+        return count == 0;
+    }
+
+    private BigDecimal parseNumber(String value) {
+        try {
+            return new BigDecimal(value);
+        } catch (NumberFormatException exception) {
+            throw new IllegalArgumentException("Invalid numeric value '" + value
+                    + "' in aggregate input column '" + column + "'", exception);
+        }
+    }
+
+    private IllegalArgumentException incompatibleState() {
+        return new IllegalArgumentException("Cannot merge incompatible aggregate states");
+    }
+}

+/**
+ * Holds the aggregate states for one immutable group key.
+ */
+final class GroupAccumulator {
+    private final List<String> groupByColumns;
+    private final List<String> key;
+    private final List<AggregateSpecification> specifications;
+    private final List<AggregateState> states;
+
+    GroupAccumulator(
+            List<String> groupByColumns,
+            List<String> key,
+            List<AggregateSpecification> specifications) {
+        this.groupByColumns = validateGroupByColumns(groupByColumns);
+        this.key = immutableNullableList(key, "Group key must not be null");
+        if (this.key.size() != this.groupByColumns.size()) {
+            throw new IllegalArgumentException("Group key size must match GROUP BY columns");
+        }
+        this.specifications = validateSpecifications(specifications, this.groupByColumns);
+        this.states = this.specifications.stream()
+                .map(AggregateStateFactory::create)
+                .toList();
+    }
+
+    List<String> key() {
+        return key;
+    }
+
+    List<AggregateSpecification> specifications() {
+        return specifications;
+    }
+
+    List<String> groupByColumns() {
+        return groupByColumns;
+    }
+
+    void add(Row row) {
+        Objects.requireNonNull(row, "Input row must not be null");
+        for (int index = 0; index < groupByColumns.size(); index++) {
+            String column = groupByColumns.get(index);
+            if (!row.values().containsKey(column)) {
+                throw new IllegalArgumentException(
+                        "Group-by column '" + column + "' is missing from row");
+            }
+            if (!Objects.equals(key.get(index), row.values().get(column))) {
+                throw new IllegalArgumentException("Input row does not match group key");
+            }
+        }
+        for (AggregateState state : states) {
+            state.add(row);
+        }
+    }
+
+    void merge(GroupAccumulator other) {
+        Objects.requireNonNull(other, "Group accumulator must not be null");
+        if (!key.equals(other.key)
+                || !groupByColumns.equals(other.groupByColumns)
+                || !specifications.equals(other.specifications)) {
+            throw new IllegalArgumentException(
+                    "Cannot merge group accumulators with incompatible keys or specifications");
+        }
+        for (int index = 0; index < states.size(); index++) {
+            states.get(index).merge(other.states.get(index));
+        }
+    }
+
+    GroupAccumulator copy() {
+        GroupAccumulator copy = new GroupAccumulator(groupByColumns, key, specifications);
+        copy.merge(this);
+        return copy;
+    }
+
+    Row result() {
+        Map<String, String> values = new LinkedHashMap<>();
+        for (int index = 0; index < groupByColumns.size(); index++) {
+            values.put(groupByColumns.get(index), key.get(index));
+        }
+        for (int index = 0; index < specifications.size(); index++) {
+            values.put(specifications.get(index).outputColumn(), states.get(index).result());
+        }
+        return new Row(values);
+    }
+
+    static void validateConfiguration(
+            List<String> groupByColumns, List<AggregateSpecification> specifications) {
+        List<String> groups = validateGroupByColumns(groupByColumns);
+        validateSpecifications(specifications, groups);
+    }
+
+    private static List<String> validateGroupByColumns(List<String> columns) {
+        Objects.requireNonNull(columns, "Group-by columns must not be null");
+        List<String> copy = new ArrayList<>(columns.size());
+        Set<String> seen = new HashSet<>();
+        for (String column : columns) {
+            if (column == null || column.isBlank()) {
+                throw new IllegalArgumentException("Group-by column must not be blank");
+            }
+            if (!seen.add(column)) {
+                throw new IllegalArgumentException("Duplicate group-by column: " + column);
+            }
+            copy.add(column);
+        }
+        return List.copyOf(copy);
+    }
+
+    private static List<AggregateSpecification> validateSpecifications(
+            List<AggregateSpecification> specifications, List<String> groupByColumns) {
+        Objects.requireNonNull(specifications, "Aggregate specifications must not be null");
+        if (specifications.isEmpty()) {
+            throw new IllegalArgumentException("At least one aggregate specification is required");
+        }
+        Set<String> outputNames = new HashSet<>(groupByColumns);
+        List<AggregateSpecification> copy = new ArrayList<>(specifications.size());
+        for (AggregateSpecification specification : specifications) {
+            Objects.requireNonNull(specification, "Aggregate specification must not be null");
+            if (!outputNames.add(specification.outputColumn())) {
+                throw new IllegalArgumentException("Duplicate aggregate output column: "
+                        + specification.outputColumn());
+            }
+            copy.add(specification);
+        }
+        return List.copyOf(copy);
+    }
+
+    private static List<String> immutableNullableList(List<String> values, String nullMessage) {
+        Objects.requireNonNull(values, nullMessage);
+        return Collections.unmodifiableList(new ArrayList<>(values));
+    }

+import io.github.payal1762005.minisql.model.Row;

+import java.util.ArrayDeque;
+import java.util.ArrayList;
+import java.util.Deque;
+import java.util.Iterator;
+import java.util.List;
+import java.util.NoSuchElementException;
+import java.util.Objects;
+import java.util.concurrent.ExecutionException;
+import java.util.concurrent.ExecutorService;
+import java.util.concurrent.Executors;
+import java.util.concurrent.Future;
+
+/**
+ * Single-consumer parallel hash aggregation. The calling thread alone pulls
+ * the child; workers aggregate bounded batches without accessing the operator.
+ */
+final class ParallelAggregateOperator implements Operator {
+    private final Operator child;
+    private final List<String> groupByColumns;
+    private final List<AggregateSpecification> specifications;
+    private final int batchSize;
+    private final int maxInFlightBatches;
+    private final ExecutorService executor;
+    private final boolean ownsExecutor;
+    private final Deque<Future<PartialAggregation>> pending = new ArrayDeque<>();
+
+    private Iterator<Row> results = List.<Row>of().iterator();
+    private boolean aggregated;
+    private boolean childClosed;
+    private boolean closed;
+    private PartialAggregation merged;
+
+    ParallelAggregateOperator(
+            Operator child,
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications,
+            int batchSize,
+            int maxInFlightBatches) {
+        this(child, groupByColumns, specifications, batchSize, maxInFlightBatches,
+                createOwnedExecutor(child, groupByColumns, specifications,
+                        batchSize, maxInFlightBatches), true);
+    }
+
+    ParallelAggregateOperator(
+            Operator child,
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications,
+            int batchSize,
+            int maxInFlightBatches,
+            ExecutorService executor) {
+        this(child, groupByColumns, specifications, batchSize, maxInFlightBatches,
+                executor, false);
+    }
+
+    ParallelAggregateOperator(
+            Operator child,
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications,
+            int batchSize,
+            int maxInFlightBatches,
+            ExecutorService executor,
+            boolean ownsExecutor) {
+        this.child = Objects.requireNonNull(child, "child operator must not be null");
+        this.groupByColumns = List.copyOf(
+                Objects.requireNonNull(groupByColumns, "Group-by columns must not be null"));
+        this.specifications = List.copyOf(
+                Objects.requireNonNull(specifications, "Aggregate specifications must not be null"));
+        GroupAccumulator.validateConfiguration(this.groupByColumns, this.specifications);
+        if (batchSize <= 0) {
+            throw new IllegalArgumentException("Batch size must be greater than zero");
+        }
+        if (maxInFlightBatches <= 0) {
+            throw new IllegalArgumentException("Maximum in-flight batches must be greater than zero");
+        }
+        this.batchSize = batchSize;
+        this.maxInFlightBatches = maxInFlightBatches;
+        this.executor = Objects.requireNonNull(executor, "Executor must not be null");
+        this.ownsExecutor = ownsExecutor;
+    }
+
+    @Override
+    public boolean hasNext() {
+        if (closed) {
+            return false;
+        }
+        try {
+            aggregateIfNeeded();
+            if (results.hasNext()) {
+                return true;
+            }
+            close();
+            return false;
+        } catch (RuntimeException | Error failure) {
+            closeAfterFailure(failure);
+            throw failure;
+        }
+    }
+
+    @Override
+    public Row next() {
+        if (!hasNext()) {
+            throw new NoSuchElementException("No aggregate result rows remain");
+        }
+        return results.next();
+    }
+
+    @Override
+    public void close() {
+        if (closed) {
+            return;
+        }
+        closed = true;
+        results = List.<Row>of().iterator();
+        pending.forEach(future -> future.cancel(true));
+        pending.clear();
+        Throwable failure = null;
+        try {
+            closeChild();
+        } catch (RuntimeException | Error closeFailure) {
+            failure = closeFailure;
+        }
+        if (ownsExecutor) {
+            try {
+                executor.shutdownNow();
+            } catch (RuntimeException | Error shutdownFailure) {
+                if (failure == null) {
+                    failure = shutdownFailure;
+                } else {
+                    failure.addSuppressed(shutdownFailure);
+                }
+            }
+        }
+        if (failure instanceof RuntimeException runtimeException) {
+            throw runtimeException;
+        }
+        if (failure instanceof Error error) {
+            throw error;
+        }
+    }
+
+    private void aggregateIfNeeded() {
+        if (aggregated) {
+            return;
+        }
+
+        try {
+            merged = new PartialAggregation(groupByColumns, specifications);
+            produceBatches();
+            closeChild();
+            while (!pending.isEmpty()) {
+                mergeNextBatch();
+            }
+            results = merged.resultRows().iterator();
+            aggregated = true;
+            merged = null;
+            shutdownOwnedExecutor();
+        } catch (RuntimeException | Error failure) {
+            closeAfterFailure(failure);
+            throw failure;
+        }
+    }
+
+    private void produceBatches() {
+        boolean exhausted = false;
+        while (!exhausted) {
+            List<Row> batch = new ArrayList<>(batchSize);
+            while (batch.size() < batchSize) {
+                if (!child.hasNext()) {
+                    exhausted = true;
+                    break;
+                }
+                batch.add(Objects.requireNonNull(
+                        child.next(), "Child operator returned a null row"));
+            }
+            if (!batch.isEmpty()) {
+                List<Row> immutableBatch = List.copyOf(batch);
+                pending.addLast(executor.submit(() ->
+                        new PartialAggregation(immutableBatch, groupByColumns, specifications)));
+            }
+            if (pending.size() >= maxInFlightBatches) {
+                mergeNextBatch();
+            }
+        }
+    }
+
+    private void mergeNextBatch() {
+        merged.merge(await(pending.removeFirst()));
+    }
+
+    private PartialAggregation await(Future<PartialAggregation> future) {
+        try {
+            return future.get();
+        } catch (InterruptedException exception) {
+            Thread.currentThread().interrupt();
+            throw new IllegalStateException("Interrupted while waiting for aggregate batch", exception);
+        } catch (ExecutionException exception) {
+            Throwable cause = exception.getCause();
+            if (cause instanceof RuntimeException runtimeException) {
+                throw runtimeException;
+            }
+            if (cause instanceof Error error) {
+                throw error;
+            }
+            throw new IllegalStateException("Aggregate batch failed", cause);
+        }
+    }
+
+    private void closeChild() {
+        if (!childClosed) {
+            childClosed = true;
+            child.close();
+        }
+    }
+
+    private void shutdownOwnedExecutor() {
+        if (ownsExecutor) {
+            executor.shutdown();
+        }
+    }
+
+    private static ExecutorService createOwnedExecutor(
+            Operator child,
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications,
+            int batchSize,
+            int maxInFlightBatches) {
+        Objects.requireNonNull(child, "child operator must not be null");
+        List<String> groups = List.copyOf(
+                Objects.requireNonNull(groupByColumns, "Group-by columns must not be null"));
+        List<AggregateSpecification> specs = List.copyOf(
+                Objects.requireNonNull(specifications, "Aggregate specifications must not be null"));

+
+import io.github.payal1762005.minisql.model.Row;
+
+import java.util.ArrayList;
+import java.util.Collection;

+import java.util.LinkedHashMap;
+import java.util.List;
+import java.util.Map;
+import java.util.Objects;
+
+/**
+ * A local insertion-ordered collection of group accumulators for a set of rows.
+ */
+final class PartialAggregation {
+    private final List<String> groupByColumns;
+    private final List<AggregateSpecification> specifications;
+    private final LinkedHashMap<List<String>, GroupAccumulator> groups = new LinkedHashMap<>();
+
+    PartialAggregation(
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications) {
+        this.groupByColumns = immutableCopy(groupByColumns, "Group-by columns must not be null");
+        this.specifications = immutableCopy(specifications,
+                "Aggregate specifications must not be null");
+        validateConfiguration();
+        if (this.groupByColumns.isEmpty()) {
+            groups.put(List.of(), new GroupAccumulator(
+                    this.groupByColumns, List.of(), this.specifications));
+        }
+    }
+
+    PartialAggregation(
+            Collection<Row> rows,
+            List<String> groupByColumns,
+            List<AggregateSpecification> specifications) {
+        this(groupByColumns, specifications);
+        Objects.requireNonNull(rows, "Input rows must not be null");
+        rows.forEach(this::add);
+    }
+
+    void add(Row row) {
+        Objects.requireNonNull(row, "Input row must not be null");
+        List<String> keyValues = new ArrayList<>(groupByColumns.size());
+        for (String column : groupByColumns) {
+            if (!row.values().containsKey(column)) {
+                throw new IllegalArgumentException(
+                        "Group-by column '" + column + "' is missing from row");
+            }
+            keyValues.add(row.values().get(column));
+        }
+        List<String> key = Collections.unmodifiableList(keyValues);
+        GroupAccumulator group = groups.computeIfAbsent(key,
+                ignored -> new GroupAccumulator(groupByColumns, key, specifications));
+        group.add(row);
+    }
+
+    void merge(PartialAggregation other) {
+        Objects.requireNonNull(other, "Partial aggregation must not be null");
+        if (!groupByColumns.equals(other.groupByColumns)
+                || !specifications.equals(other.specifications)) {
+            throw new IllegalArgumentException(
+                    "Cannot merge partial aggregations with incompatible specifications");
+        }
+        for (Map.Entry<List<String>, GroupAccumulator> entry : other.groups.entrySet()) {
+            GroupAccumulator current = groups.get(entry.getKey());
+            if (current == null) {
+                groups.put(entry.getKey(), entry.getValue().copy());
+            } else {
+                current.merge(entry.getValue());
+            }
+        }
+    }
+
+    List<Row> resultRows() {
+        return groups.values().stream().map(GroupAccumulator::result).toList();
+    }
+
+    private void validateConfiguration() {
+        GroupAccumulator.validateConfiguration(groupByColumns, specifications);
+    }
+
+    private static <T> List<T> immutableCopy(List<T> values, String nullMessage) {
+        Objects.requireNonNull(values, nullMessage);
+        return List.copyOf(values);
+    }

+    private static class TrackingOperator implements Operator {
+        private final List<Row> rows;
+        final AtomicInteger closeCount = new AtomicInteger();
+        private int index;
+
+        private TrackingOperator(List<Row> rows) {
+            this.rows = rows;
+        }
+
+        @Override
+        public boolean hasNext() {
+            return index < rows.size();
+        }
+
+        @Override
+        public Row next() {
+            if (!hasNext()) {
+                throw new NoSuchElementException();
+            }
+            return rows.get(index++);
+        }
+
+        @Override
+        public void close() {
+            closeCount.incrementAndGet();
+        }
+    }
+
+    private static final class ThreadCheckingOperator extends TrackingOperator {
+        private Thread accessThread;
+
+        private ThreadCheckingOperator(List<Row> rows) {
+            super(rows);
+        }
+
+        @Override
+        public boolean hasNext() {
+            assertCallerThread();
+            return super.hasNext();
+        }
+
+        @Override
+        public Row next() {
+            assertCallerThread();
+            return super.next();
+        }
+
+        @Override
+        public void close() {
+            assertCallerThread();
+            super.close();
+        }
+
+        private void assertCallerThread() {
+            if (accessThread == null) {
+                accessThread = Thread.currentThread();
+            }
+            if (accessThread != Thread.currentThread()) {
+                throw new AssertionError("Child operator accessed from a worker thread");
+            }
+        }
+    }
+
+    private static class TrackingExecutor extends AbstractExecutorService {
+        private final ExecutorService delegate;
+        private final AtomicInteger outstanding = new AtomicInteger();
+        private final AtomicInteger maximumOutstanding = new AtomicInteger();
+
+        private TrackingExecutor(int threads) {
+            delegate = Executors.newFixedThreadPool(threads);
+        }
+
+        @Override
+        public void execute(Runnable command) {
+            int active = outstanding.incrementAndGet();
+            maximumOutstanding.accumulateAndGet(active, Math::max);
+            delegate.execute(() -> {
+                try {
+                    command.run();
+                } finally {
+                    outstanding.decrementAndGet();
+                }
+            });
+        }
+
+        @Override
+        public void shutdown() {
+            delegate.shutdown();
+        }
+
+        @Override
+        public List<Runnable> shutdownNow() {
+            return delegate.shutdownNow();
+        }
+
+        @Override
+        public boolean isShutdown() {
+            return delegate.isShutdown();
+        }
+
+        @Override
+        public boolean isTerminated() {
+            return delegate.isTerminated();
+        }
+
+        @Override
+        public boolean awaitTermination(long timeout, TimeUnit unit) throws InterruptedException {
+            return delegate.awaitTermination(timeout, unit);
+        }
+
+        private int maximumOutstanding() {
+            return maximumOutstanding.get();
+        }
+    }
+
+    private static final class OutOfOrderExecutor extends TrackingExecutor {
+        private final AtomicInteger taskNumber = new AtomicInteger();
+        private final CountDownLatch secondCompleted = new CountDownLatch(1);
+
+        private OutOfOrderExecutor() {
+            super(2);
+        }
+
+        @Override
+        public void execute(Runnable command) {
+            int task = taskNumber.incrementAndGet();
+            super.execute(() -> {
+                if (task == 1) {
+                    try {
+                        if (!secondCompleted.await(5, TimeUnit.SECONDS)) {
+                            throw new IllegalStateException("Second batch did not complete");
+                        }
+                    } catch (InterruptedException exception) {
+                        Thread.currentThread().interrupt();
+                        throw new IllegalStateException(exception);
+                    }
+                    command.run();
+                } else {
+                    command.run();
+                    secondCompleted.countDown();
+                }
+            });
+        }
+    }
